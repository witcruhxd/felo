<script>
const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');
let w, h, dpr, cx, cy;

function resize() {
  dpr = window.devicePixelRatio || 1;
  w = window.innerWidth;
  h = window.innerHeight;
  canvas.width = w * dpr;
  canvas.height = h * dpr;
  canvas.style.width = w + 'px';
  canvas.style.height = h + 'px';
  ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  cx = w / 2;
  cy = h / 2;
}
window.addEventListener('resize', resize);
resize();

// Sadece doğum günü temalı kelimeler
const words = [
  "İYİ Kİ DOĞDUN",
  "NİCE MUTLU YILLARA",
  "DOĞUM GÜNÜN KUTLU OLSUN",
  "MUTLU YILLAR",
  "NİCE YAŞLARA",
  "İYİ Kİ VARSIN",
  "DİLEK TUT",
  "MUM ÜFLE",
  "PASTA",
  "KUTLU OLSUN",
  "DOĞUM GÜNÜ",
  "SAĞLIK",
  "MUTLULUK",
  "BAŞARI",
  "SEVGİLERLE",
  "DOĞUM GÜNÜN",
  "HAPPY BIRTHDAY",
  "FELONA",
  "HEDİYELER",
  "KUTLAMALAR",
  "NİCE SENELERE",
  "DOĞUM GÜNÜN KUTLU",
  "İYİ Kİ DOĞDUN FELONA"
];

function heartX(t) { return 16 * Math.pow(Math.sin(t), 3); }
function heartY(t) { return -(13 * Math.cos(t) - 5 * Math.cos(2*t) - 2 * Math.cos(3*t) - Math.cos(4*t)); }

let rotation = 0;
let rotationSpeed = 0.0008;
let isDragging = false;
let lastX = 0;
let velocity = 0;

canvas.addEventListener('mousedown', e => {
  isDragging = true; lastX = e.clientX; velocity = 0;
});
canvas.addEventListener('mousemove', e => {
  if (isDragging) {
    const dx = e.clientX - lastX;
    rotation += dx * 0.005;
    velocity = dx * 0.005;
    lastX = e.clientX;
  }
});
canvas.addEventListener('mouseup', () => { isDragging = false; });
canvas.addEventListener('mouseleave', () => { isDragging = false; });

canvas.addEventListener('touchstart', e => {
  isDragging = true; lastX = e.touches[0].clientX; velocity = 0;
}, {passive: true});
canvas.addEventListener('touchmove', e => {
  if (isDragging) {
    const dx = e.touches[0].clientX - lastX;
    rotation += dx * 0.005;
    velocity = dx * 0.005;
    lastX = e.touches[0].clientX;
  }
}, {passive: true});
canvas.addEventListener('touchend', () => { isDragging = false; });

class Particle {
  constructor() { this.init(); }
  init() {
    this.angle = Math.random() * Math.PI * 2;
    this.radiusFactor = 0.95 + Math.random() * 0.5;
    this.speed = (0.0005 + Math.random() * 0.001) * (Math.random() > 0.5 ? 1 : -1);
    this.word = words[Math.floor(Math.random() * words.length)];
    this.size = 8 + Math.random() * 7;
    this.baseAlpha = 0.35 + Math.random() * 0.6;
    this.pulseOffset = Math.random() * Math.PI * 2;
    this.hue = 335 + Math.random() * 25;
    this.z = 0.85 + Math.random() * 0.4;
  }
  draw(time) {
    const beat = 1 + 0.07 * Math.sin(time * 0.005) + 0.03 * Math.sin(time * 0.01);
    const scale = Math.min(w, h) * 0.028 * this.radiusFactor * this.z * beat;

    this.angle += this.speed;
    const a = this.angle + rotation;

    const x = cx + heartX(a) * scale;
    const y = cy + heartY(a) * scale;

    const alpha = this.baseAlpha * (0.6 + 0.4 * Math.sin(time * 0.003 + this.pulseOffset));

    ctx.save();
    ctx.translate(x, y);
    ctx.font = `600 ${this.size}px 'Segoe UI', sans-serif`;
    ctx.fillStyle = `hsla(${this.hue}, 100%, 75%, ${alpha})`;
    ctx.shadowColor = `hsla(${this.hue}, 100%, 55%, ${alpha})`;
    ctx.shadowBlur = 12;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillText(this.word, 0, 0);
    ctx.restore();
  }
}

const particles = [];
const PARTICLE_COUNT = 100;
for (let i = 0; i < PARTICLE_COUNT; i++) particles.push(new Particle());

const stars = [];
for (let i = 0; i < 150; i++) {
  stars.push({
    x: Math.random() * w,
    y: Math.random() * h,
    r: Math.random() * 1.4,
    a: Math.random(),
    s: 0.005 + Math.random() * 0.02
  });
}

function drawStars(time) {
  stars.forEach(s => {
    s.a = 0.2 + 0.8 * Math.abs(Math.sin(time * s.s));
    ctx.beginPath();
    ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
    ctx.fillStyle = `rgba(255, 200, 220, ${s.a * 0.5})`;
    ctx.fill();
  });
}

function drawCenterText(time) {
  const beat = 1 + 0.05 * Math.sin(time * 0.005) + 0.025 * Math.sin(time * 0.01);
  const baseSize = Math.min(w, h) * 0.085;

  ctx.save();
  ctx.translate(cx, cy);
  ctx.scale(beat, beat);
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';

  const ustSize = baseSize * 0.42;
  ctx.font = `500 ${ustSize}px 'Segoe UI', sans-serif`;
  ctx.fillStyle = 'rgba(255, 130, 170, 0.85)';
  ctx.shadowColor = '#ff0044';
  ctx.shadowBlur = 22;
  ctx.fillText('İYİ Kİ DOĞDUN', 0, -baseSize * 0.45);

  ctx.font = `900 ${baseSize}px 'Segoe UI', sans-serif`;
  const grad = ctx.createLinearGradient(0, -baseSize * 0.5, 0, baseSize * 0.5);
  grad.addColorStop(0, '#ffffff');
  grad.addColorStop(0.5, '#ff6699');
  grad.addColorStop(1, '#ff0044');
  ctx.fillStyle = grad;
  ctx.shadowColor = '#ff0044';
  ctx.shadowBlur = 35;
  ctx.fillText('FELONA', 0, 0);

  ctx.restore();
}

function drawHeartOutline(time) {
  const beat = 1 + 0.07 * Math.sin(time * 0.005) + 0.03 * Math.sin(time * 0.01);
  const scale = Math.min(w, h) * 0.028 * beat;

  ctx.save();
  ctx.translate(cx, cy);
  ctx.beginPath();
  for (let t = 0; t <= Math.PI * 2 + 0.01; t += 0.008) {
    const x = heartX(t) * scale;
    const y = heartY(t) * scale;
    if (t === 0) ctx.moveTo(x, y);
    else ctx.lineTo(x, y);
  }
  ctx.closePath();

  ctx.shadowColor = '#ff0044';
  ctx.shadowBlur = 25;
  ctx.strokeStyle = 'rgba(255, 60, 120, 0.35)';
  ctx.lineWidth = 1.5;
  ctx.stroke();

  ctx.shadowBlur = 50;
  ctx.strokeStyle = 'rgba(255, 0, 68, 0.15)';
  ctx.lineWidth = 5;
  ctx.stroke();
  ctx.restore();
}

const floaters = [];
for (let i = 0; i < 20; i++) {
  floaters.push({
    x: Math.random() * w,
    y: h + Math.random() * h,
    s: 6 + Math.random() * 8,
    v: 0.3 + Math.random() * 0.6,
    a: 0.2 + Math.random() * 0.3,
    drift: (Math.random() - 0.5) * 0.5
  });
}
function drawFloaters() {
  floaters.forEach(f => {
    f.y -= f.v;
    f.x += f.drift;
    if (f.y < -20) { f.y = h + 20; f.x = Math.random() * w; }
    ctx.save();
    ctx.translate(f.x, f.y);
    ctx.font = `${f.s * 2}px serif`;
    ctx.fillStyle = `rgba(255, 80, 130, ${f.a})`;
    ctx.shadowColor = '#ff0044';
    ctx.shadowBlur = 12;
    ctx.textAlign = 'center';
    ctx.fillText('♥', 0, 0);
    ctx.restore();
  });
}

const rings = [];
let lastRing = 0;
function spawnRing(time) {
  if (time - lastRing > 1400) {
    lastRing = time;
    rings.push({ r: 0, alpha: 0.5 });
  }
}
function drawRings() {
  for (let i = rings.length - 1; i >= 0; i--) {
    const ring = rings[i];
    ring.r += 2;
    ring.alpha -= 0.004;
    if (ring.alpha <= 0) { rings.splice(i, 1); continue; }
    ctx.save();
    ctx.translate(cx, cy);
    ctx.beginPath();
    for (let t = 0; t <= Math.PI * 2 + 0.01; t += 0.02) {
      const x = heartX(t) * (Math.min(w, h) * 0.028) * (1 + ring.r / 300);
      const y = heartY(t) * (Math.min(w, h) * 0.028) * (1 + ring.r / 300);
      if (t === 0) ctx.moveTo(x, y);
      else ctx.lineTo(x, y);
    }
    ctx.closePath();
    ctx.strokeStyle = `rgba(255, 60, 120, ${ring.alpha})`;
    ctx.lineWidth = 1;
    ctx.shadowColor = '#ff0044';
    ctx.shadowBlur = 20;
    ctx.stroke();
    ctx.restore();
  }
}

function animate(time) {
  ctx.fillStyle = 'rgba(10, 0, 4, 0.28)';
  ctx.fillRect(0, 0, w, h);

  drawStars(time);
  drawRings();
  drawHeartOutline(time);
  drawFloaters();

  particles.forEach(p => p.draw(time));

  drawCenterText(time);

  if (!isDragging) {
    rotation += rotationSpeed + velocity;
    velocity *= 0.95;
    if (Math.abs(velocity) < 0.0001) velocity = 0;
  }

  spawnRing(time);
  requestAnimationFrame(animate);
}
requestAnimationFrame(animate);
